Option Explicit

'====================================================================
' 現行 / 刷新後 データ高速比較マクロ
'
' 【基本設定】
' ・データ開始行：3行目
' ・Key：A列
' ・比較対象シート：現行、刷新後
'
' 【第1グループ】
' 刷新後 H:V
'       ↓
' 現行 K:Y
'
' 【第2グループ】
' 刷新後 W:AH
'       ↓
' 現行 Z:AK + AL:AW
'
' 【比較結果】
' 刷新後 AX:BX
'
' 【Key状態】
' 刷新後 AV
' 現行 BP
'
' 【比率比較】
' H ↔ K
' I ↔ L
' M ↔ P
' N ↔ Q
' R ↔ U
' S ↔ V
'
' 上記6項目は小数点以下11桁まで比較する。
'
' 【その他】
' その他の項目は通常の数値比較を行う。
'
' 【数値以外の処理】
' 空白、文字列、Excelエラー値は0として扱う。
'
' 【色設定】
' 比較結果がFALSEの場合：
'   → 背景色を薄い赤色にする
'
' 「刷新後のみ」「現行のみ」の場合：
'   → 赤文字＋太字にする
'
' 大量データに対応するため、
' Array + Dictionary + 条件付き書式を使用する。
'====================================================================

Public Sub データ比較_高速版()

    Const DATA_START_ROW As Long = 3
    Const RESULT_START_COL As Long = 50   ' AX
    Const RESULT_COUNT As Long = 27

    Dim wsOld As Worksheet
    Dim wsNew As Worksheet

    Set wsOld = ThisWorkbook.Worksheets("現行")
    Set wsNew = ThisWorkbook.Worksheets("刷新後")

    Dim lastRowOld As Long
    Dim lastRowNew As Long

    lastRowOld = wsOld.Cells(wsOld.Rows.Count, "A").End(xlUp).Row
    lastRowNew = wsNew.Cells(wsNew.Rows.Count, "A").End(xlUp).Row

    Dim oldData As Variant
    Dim newData As Variant
    Dim resultData As Variant
    Dim statusNew As Variant
    Dim statusOld As Variant

    Dim dictOld As Object
    Dim dictNew As Object

    Dim r As Long
    Dim c As Long
    Dim oldRow As Long

    Dim key As String
    Dim newValue As Double
    Dim oldValue As Double

    Dim prevCalculation As XlCalculation
    Dim prevScreenUpdating As Boolean
    Dim prevEnableEvents As Boolean

    prevCalculation = Application.Calculation
    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents

    On Error GoTo ErrorHandler

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    ' 前回の結果・色・条件付き書式をクリア
    If lastRowNew >= DATA_START_ROW Then
        With wsNew.Range("AV" & DATA_START_ROW & ":BX" & lastRowNew)
            .ClearContents
            .Interior.Pattern = xlNone
            .Font.ColorIndex = xlAutomatic
            .Font.Bold = False
            .FormatConditions.Delete
        End With
    End If

    If lastRowOld >= DATA_START_ROW Then
        With wsOld.Range("BP" & DATA_START_ROW & ":BP" & lastRowOld)
            .ClearContents
            .Font.ColorIndex = xlAutomatic
            .Font.Bold = False
            .FormatConditions.Delete
        End With
    End If

    ' データを一括で配列に読み込む
    If lastRowOld >= DATA_START_ROW Then
        oldData = wsOld.Range("A" & DATA_START_ROW & ":AW" & lastRowOld).Value2
    End If

    If lastRowNew >= DATA_START_ROW Then
        newData = wsNew.Range("A" & DATA_START_ROW & ":AH" & lastRowNew).Value2
    End If

    ' 現行のKeyをDictionaryに登録
    Set dictOld = CreateObject("Scripting.Dictionary")
    dictOld.CompareMode = vbTextCompare

    If lastRowOld >= DATA_START_ROW Then
        For r = 1 To UBound(oldData, 1)
            key = GetKey(oldData(r, 1))
            If Len(key) > 0 Then
                dictOld(key) = r
            End If
        Next r
    End If

    ' 刷新後のKeyをDictionaryに登録
    Set dictNew = CreateObject("Scripting.Dictionary")
    dictNew.CompareMode = vbTextCompare

    If lastRowNew >= DATA_START_ROW Then
        For r = 1 To UBound(newData, 1)
            key = GetKey(newData(r, 1))
            If Len(key) > 0 Then
                dictNew(key) = r
            End If
        Next r
    End If

    ' 比較処理
    If lastRowNew >= DATA_START_ROW Then

        ReDim resultData(1 To UBound(newData, 1), 1 To RESULT_COUNT)

        For r = 1 To UBound(newData, 1)

            key = GetKey(newData(r, 1))

            If Len(key) > 0 Then

                If dictOld.Exists(key) Then

                    oldRow = dictOld(key)

                    ' 第1グループ：刷新後 H:V ↔ 現行 K:Y
                    For c = 0 To 14

                        newValue = ToNumberFast(newData(r, 8 + c))
                        oldValue = ToNumberFast(oldData(oldRow, 11 + c))

                        ' 比率項目：H/K、I/L、M/P、N/Q、R/U、S/V
                        If c = 0 Or c = 1 Or c = 5 Or c = 6 Or c = 10 Or c = 11 Then
                            resultData(r, c + 1) = _
                                (Round(newValue, 11) = Round(oldValue, 11))
                        Else
                            resultData(r, c + 1) = _
                                (newValue = oldValue)
                        End If

                    Next c

                    ' 第2グループ：刷新後 W:AH ↔ 現行2列の合計
                    For c = 0 To 11

                        newValue = ToNumberFast(newData(r, 23 + c))

                        oldValue = _
                            ToNumberFast(oldData(oldRow, 26 + c)) + _
                            ToNumberFast(oldData(oldRow, 38 + c))

                        resultData(r, 16 + c) = _
                            (newValue = oldValue)

                    Next c

                Else

                    ' Keyが刷新後にのみ存在する場合
                    For c = 1 To RESULT_COUNT
                        resultData(r, c) = False
                    Next c

                End If

            End If

        Next r

        ' 比較結果を一括で書き込む
        wsNew.Range("AX" & DATA_START_ROW & ":BX" & lastRowNew).Value = resultData

        ' 刷新後のみのKey
        ReDim statusNew(1 To UBound(newData, 1), 1 To 1)

        For r = 1 To UBound(newData, 1)

            key = GetKey(newData(r, 1))

            If Len(key) > 0 Then
                If Not dictOld.Exists(key) Then
                    statusNew(r, 1) = "刷新後のみ"
                End If
            End If

        Next r

        wsNew.Range("AV" & DATA_START_ROW & ":AV" & lastRowNew).Value = statusNew

    End If

    ' 現行のみのKey
    If lastRowOld >= DATA_START_ROW Then

        ReDim statusOld(1 To UBound(oldData, 1), 1 To 1)

        For r = 1 To UBound(oldData, 1)

            key = GetKey(oldData(r, 1))

            If Len(key) > 0 Then
                If Not dictNew.Exists(key) Then
                    statusOld(r, 1) = "現行のみ"
                End If
            End If

        Next r

        wsOld.Range("BP" & DATA_START_ROW & ":BP" & lastRowOld).Value = statusOld

    End If

    ' 比較結果ヘッダー：第1グループ
    For c = 0 To 14

        wsNew.Cells(1, RESULT_START_COL + c).Value = _
            wsNew.Cells(1, 8 + c).Value

        wsNew.Cells(2, RESULT_START_COL + c).Value = _
            "現行 " & ColumnLetter(11 + c)

    Next c

    ' 比較結果ヘッダー：第2グループ
    For c = 0 To 11

        wsNew.Cells(1, RESULT_START_COL + 15 + c).Value = _
            wsNew.Cells(1, 23 + c).Value

        wsNew.Cells(2, RESULT_START_COL + 15 + c).Value = _
            "現行 " & ColumnLetter(26 + c) & _
            " + " & ColumnLetter(38 + c)

    Next c

    ' Key状態ヘッダー
    wsNew.Range("AV1").Value = "Key状態"
    wsNew.Range("AV2").Value = "刷新後のみ"

    wsOld.Range("BP1").Value = "Key状態"
    wsOld.Range("BP2").Value = "現行のみ"

    wsNew.Range("AV1:BX2").Font.Bold = True
    wsOld.Range("BP1:BP2").Font.Bold = True

    ' FALSEのセルを薄い赤色にする
    If lastRowNew >= DATA_START_ROW Then

        Dim rngResult As Range
        Dim fcFalse As FormatCondition

        Set rngResult = wsNew.Range("AX" & DATA_START_ROW & ":BX" & lastRowNew)

        Set fcFalse = rngResult.FormatConditions.Add( _
            Type:=xlExpression, _
            Formula1:="=AND(ISLOGICAL(AX3),AX3=FALSE)" _
        )

        fcFalse.Interior.Color = RGB(255, 199, 206)

    End If

    ' 「刷新後のみ」を赤文字＋太字にする
    If lastRowNew >= DATA_START_ROW Then

        Dim rngStatusNew As Range
        Dim fcStatusNew As FormatCondition

        Set rngStatusNew = wsNew.Range( _
            "AV" & DATA_START_ROW & ":AV" & lastRowNew _
        )

        Set fcStatusNew = rngStatusNew.FormatConditions.Add( _
            Type:=xlExpression, _
            Formula1:="=$AV3=""刷新後のみ""" _
        )

        fcStatusNew.Font.Color = RGB(255, 0, 0)
        fcStatusNew.Font.Bold = True

    End If

    ' 「現行のみ」を赤文字＋太字にする
    If lastRowOld >= DATA_START_ROW Then

        Dim rngStatusOld As Range
        Dim fcStatusOld As FormatCondition

        Set rngStatusOld = wsOld.Range( _
            "BP" & DATA_START_ROW & ":BP" & lastRowOld _
        )

        Set fcStatusOld = rngStatusOld.FormatConditions.Add( _
            Type:=xlExpression, _
            Formula1:="=$BP3=""現行のみ""" _
        )

        fcStatusOld.Font.Color = RGB(255, 0, 0)
        fcStatusOld.Font.Bold = True

    End If

    ' Excelの設定を元に戻す
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalculation

    MsgBox _
        "データ比較が完了しました。" & vbCrLf & vbCrLf & _
        "比較結果：" & vbCrLf & _
        "刷新後 AX:BX" & vbCrLf & vbCrLf & _
        "Key状態：" & vbCrLf & _
        "刷新後のみ → 刷新後 AV" & vbCrLf & _
        "現行のみ → 現行 BP" & vbCrLf & vbCrLf & _
        "FALSE → 薄い赤色", _
        vbInformation, _
        "データ比較完了"

    Exit Sub

ErrorHandler:

    ' エラーが発生してもExcelの設定を元に戻す
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalculation

    MsgBox _
        "データ比較中にエラーが発生しました。" & vbCrLf & vbCrLf & _
        "エラー番号：" & Err.Number & vbCrLf & _
        "エラー内容：" & Err.Description, _
        vbCritical, _
        "データ比較エラー"

End Sub


'====================================================================
' Key取得処理
'
' 空白やエラー値の場合は空文字を返す。
' それ以外は文字列に変換し、前後の空白を削除する。
'====================================================================

Private Function GetKey(ByVal v As Variant) As String

    If IsError(v) Then
        GetKey = ""
    ElseIf IsEmpty(v) Then
        GetKey = ""
    Else
        GetKey = Trim$(CStr(v))
    End If

End Function


'====================================================================
' 数値変換処理
'
' 空白      → 0
' 数値以外  → 0
' エラー値  → 0
' 数値      → Double
'====================================================================

Private Function ToNumberFast(ByVal v As Variant) As Double

    If IsError(v) Then
        ToNumberFast = 0
    ElseIf IsEmpty(v) Then
        ToNumberFast = 0
    ElseIf IsNumeric(v) Then
        ToNumberFast = CDbl(v)
    Else
        ToNumberFast = 0
    End If

End Function


'====================================================================
' 列番号を列記号に変換
'
' 例：
' 11 → K
' 26 → Z
' 38 → AL
' 50 → AX
'====================================================================

Private Function ColumnLetter(ByVal colNum As Long) As String

    ColumnLetter = _
        Split( _
            Application.Cells(1, colNum).Address(True, False), _
            "$" _
        )(0)

End Function
