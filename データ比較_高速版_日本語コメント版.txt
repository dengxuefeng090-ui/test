Option Explicit
'====================================================================
' 現行 / 刷新後 データ高速比較
'
' データ開始行：3行目
' Key：A列
'
' 第1グループ：
'   刷新後 H:V
'   ↓
'   現行 K:Y
'
' 第2グループ：
'   刷新後 W:AH
'   ↓
'   現行 Z:AK + AL:AW
'
' 比較結果：
'   刷新後 AX:BX
'
' Key状態：
'   刷新後 AV
'   現行 BP
'
' 特殊比率比較：
'   H ↔ K
'   I ↔ L
'   M ↔ P
'   N ↔ Q
'   R ↔ U
'   S ↔ V
'
' 上記6グループ：
'   小数点以下11桁で比較
'
' その他の項目：
'   通常の数値比較
'
' 空白 / 数値以外 / Error：
'   0として処理
'====================================================================
Public Sub データ比較_高速版()
    Const DATA_START_ROW As Long = 3
    Const RESULT_START_COL As Long = 50   ' AX
    Const RESULT_COUNT As Long = 27
    Dim wsOld As Worksheet
    Dim wsNew As Worksheet
    Dim lastRowOld As Long
    Dim lastRowNew As Long
    Dim oldData As Variant
    Dim newData As Variant
    Dim resultData As Variant
    Dim statusNew As Variant
    Dim statusOld As Variant
    Dim dictOld As Object
    Dim dictNew As Object
    Dim r As Long
    Dim c As Long
    Dim oldRow As Long
    Dim key As String
    Dim newValue As Double
    Dim oldValue As Double
    '------------------------------------------------------------
    ' Excelの元の状態を保存
    '------------------------------------------------------------
    Dim prevCalculation As XlCalculation
    Dim prevScreenUpdating As Boolean
    Dim prevEnableEvents As Boolean
    '------------------------------------------------------------
    ' ワークシート
    '------------------------------------------------------------
    Set wsOld = ThisWorkbook.Worksheets("現行")
    Set wsNew = ThisWorkbook.Worksheets("刷新後")
    '------------------------------------------------------------
    ' 最終行を取得
    ' KeyはA列
    '------------------------------------------------------------
    lastRowOld = wsOld.Cells(wsOld.Rows.Count, "A").End(xlUp).Row
    lastRowNew = wsNew.Cells(wsNew.Rows.Count, "A").End(xlUp).Row
    '------------------------------------------------------------
    ' Excelの現在の設定を保存
    '------------------------------------------------------------
    prevCalculation = Application.Calculation
    prevScreenUpdating = Application.ScreenUpdating
    prevEnableEvents = Application.EnableEvents
    On Error GoTo ErrorHandler
    '============================================================
    ' Excelの画面更新を停止して高速化
    '============================================================
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    '============================================================
    ' 旧結果をクリア
    '
    ' 刷新後：
    '   AV:BX
    '
    ' 現行：
    '   BP
    '============================================================
    If lastRowNew >= DATA_START_ROW Then
        wsNew.Range( _
            "AV" & DATA_START_ROW & ":BX" & lastRowNew _
        ).ClearContents
    End If
    If lastRowOld >= DATA_START_ROW Then
        wsOld.Range( _
            "BP" & DATA_START_ROW & ":BP" & lastRowOld _
        ).ClearContents
    End If
    '============================================================
    ' データを一括読み込み
    '
    ' 現行：
    '   A:AW
    '
    ' 刷新後：
    '   A:AH
    '============================================================
    If lastRowOld >= DATA_START_ROW Then
        oldData = wsOld.Range( _
            "A" & DATA_START_ROW & ":AW" & lastRowOld _
        ).Value2
    End If
    If lastRowNew >= DATA_START_ROW Then
        newData = wsNew.Range( _
            "A" & DATA_START_ROW & ":AH" & lastRowNew _
        ).Value2
    End If
    '============================================================
    ' 現行Key Dictionaryを作成
    '============================================================
    Set dictOld = CreateObject("Scripting.Dictionary")
    ' 大文字・小文字を区別しない
    dictOld.CompareMode = vbTextCompare
    If lastRowOld >= DATA_START_ROW Then
        For r = 1 To UBound(oldData, 1)
            key = GetKey(oldData(r, 1))
            If Len(key) > 0 Then
                dictOld(key) = r
            End If
        Next r
    End If
    '============================================================
    ' 刷新後Key Dictionaryを作成
    '============================================================
    Set dictNew = CreateObject("Scripting.Dictionary")
    dictNew.CompareMode = vbTextCompare
    If lastRowNew >= DATA_START_ROW Then
        For r = 1 To UBound(newData, 1)
            key = GetKey(newData(r, 1))
            If Len(key) > 0 Then
                dictNew(key) = r
            End If
        Next r
    End If
    '============================================================
    ' 比較結果配列を作成
    '
    ' AX:BX = 27列
    '============================================================
    If lastRowNew >= DATA_START_ROW Then
        ReDim resultData( _
            1 To UBound(newData, 1), _
            1 To RESULT_COUNT _
        )
        '========================================================
        ' 比較開始
        '========================================================
        For r = 1 To UBound(newData, 1)
            key = GetKey(newData(r, 1))
            If Len(key) > 0 Then
                '------------------------------------------------
                ' Keyが現行に存在
                '------------------------------------------------
                If dictOld.Exists(key) Then
                    oldRow = dictOld(key)
                    '================================================
                    ' 第1グループ
                    '
                    ' 刷新後 H:V
                    ' ↓
                    ' 現行 K:Y
                    '
                    ' 全15項目
                    '================================================
                    For c = 0 To 14
                        newValue = ToNumberFast( _
                            newData(r, 8 + c) _
                        )
                        oldValue = ToNumberFast( _
                            oldData(oldRow, 11 + c) _
                        )
                        '------------------------------------------------
                        ' 比率型：
                        '
                        ' c = 0  → H / K
                        ' c = 1  → I / L
                        ' c = 5  → M / P
                        ' c = 6  → N / Q
                        ' c = 10 → R / U
                        ' c = 11 → S / V
                        '
                        ' 小数点以下11桁で比較
                        '------------------------------------------------
                        Select Case c
                            Case 0, 1, 5, 6, 10, 11
                                resultData(r, c + 1) = _
                                    (Round(newValue, 11) = _
                                     Round(oldValue, 11))
                            '------------------------------------------------
                            ' 通常の数値型
                            '------------------------------------------------
                            Case Else
                                resultData(r, c + 1) = _
                                    (newValue = oldValue)
                        End Select
                    Next c
                    '================================================
                    ' 第2グループ
                    '
                    ' 刷新後 W:AH
                    '
                    ' 現行：
                    ' Z  + AL
                    ' AA + AM
                    ' AB + AN
                    ' ...
                    ' AK + AW
                    '
                    ' 全12項目
                    '================================================
                    For c = 0 To 11
                        newValue = ToNumberFast( _
                            newData(r, 23 + c) _
                        )
                        oldValue = _
                            ToNumberFast( _
                                oldData(oldRow, 26 + c) _
                            ) _
                            + _
                            ToNumberFast( _
                                oldData(oldRow, 38 + c) _
                            )
                        ' 通常の数値比較
                        resultData(r, 16 + c) = _
                            (newValue = oldValue)
                    Next c
                Else
                    '================================================
                    ' Keyが刷新後にのみ存在
                    '
                    ' AX:BXをすべてFalseに設定
                    '
                    ' AVには別途書き込み：
                    ' 刷新後のみ
                    '================================================
                    For c = 1 To RESULT_COUNT
                        resultData(r, c) = False
                    Next c
                End If
            End If
        Next r
        '========================================================
        ' 比較結果を一括書き込み
        '========================================================
        wsNew.Range( _
            "AX" & DATA_START_ROW & ":BX" & lastRowNew _
        ).Value = resultData
        '========================================================
        ' 刷新後のKey状態を書き込み
        '
        ' 刷新後に存在
        ' 現行に存在しない
        '
        ' → AV
        '========================================================
        ReDim statusNew( _
            1 To UBound(newData, 1), _
            1 To 1 _
        )
        For r = 1 To UBound(newData, 1)
            key = GetKey(newData(r, 1))
            If Len(key) > 0 Then
                If Not dictOld.Exists(key) Then
                    statusNew(r, 1) = "刷新後のみ"
                End If
            End If
        Next r
        wsNew.Range( _
            "AV" & DATA_START_ROW & ":AV" & lastRowNew _
        ).Value = statusNew
    End If
    '============================================================
    ' 現行のKey状態を書き込み
    '
    ' 現行に存在
    ' 刷新後に存在しない
    '
    ' → BP
    '============================================================
    If lastRowOld >= DATA_START_ROW Then
        ReDim statusOld( _
            1 To UBound(oldData, 1), _
            1 To 1 _
        )
        For r = 1 To UBound(oldData, 1)
            key = GetKey(oldData(r, 1))
            If Len(key) > 0 Then
                If Not dictNew.Exists(key) Then
                    statusOld(r, 1) = "現行のみ"
                End If
            End If
        Next r
        wsOld.Range( _
            "BP" & DATA_START_ROW & ":BP" & lastRowOld _
        ).Value = statusOld
    End If
    '============================================================
    ' 結果タイトルを設定
    '============================================================
    '------------------------------------------------------------
    ' 第1グループ
    '
    ' AX:BL
    '
    ' 刷新後 H:V
    ' 現行 K:Y
    '------------------------------------------------------------
    For c = 0 To 14
        ' 1行目：
        ' 刷新後の対応項目の物理名称
        wsNew.Cells( _
            1, _
            RESULT_START_COL + c _
        ).Value = _
            wsNew.Cells(1, 8 + c).Value
        ' 2行目：
        ' 現行の対応列
        wsNew.Cells( _
            2, _
            RESULT_START_COL + c _
        ).Value = _
            "現行 " & ColumnLetter(11 + c)
    Next c
    '------------------------------------------------------------
    ' 第2グループ
    '
    ' BM:BX
    '
    ' 刷新後 W:AH
    ' 現行 Z:AK + AL:AW
    '------------------------------------------------------------
    For c = 0 To 11
        ' 1行目：
        ' 刷新後の対応項目の物理名称
        wsNew.Cells( _
            1, _
            RESULT_START_COL + 15 + c _
        ).Value = _
            wsNew.Cells(1, 23 + c).Value
        ' 2行目：
        ' 現行の対応する2列
        wsNew.Cells( _
            2, _
            RESULT_START_COL + 15 + c _
        ).Value = _
            "現行 " & _
            ColumnLetter(26 + c) & _
            " + " & _
            ColumnLetter(38 + c)
    Next c
    '============================================================
    ' Key状态标题
    '============================================================
    wsNew.Range("AV1").Value = "Key状態"
    wsNew.Range("AV2").Value = "刷新後のみ"
    wsOld.Range("BP1").Value = "Key状態"
    wsOld.Range("BP2").Value = "現行のみ"
    '============================================================
    ' タイトルを太字に設定
    '============================================================
    wsNew.Range("AV1:BX2").Font.Bold = True
    wsOld.Range("BP1:BP2").Font.Bold = True
    '============================================================
    ' Excelの元の設定に戻す
    '============================================================
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalculation
    '============================================================
    ' 完了メッセージ
    '============================================================
    MsgBox _
        "データ比較が完了しました！" & vbCrLf & vbCrLf & _
        "比較結果：" & vbCrLf & _
        "刷新後 AX:BX" & vbCrLf & vbCrLf & _
        "Key状態：" & vbCrLf & _
        "刷新後のみ → 刷新後 AV" & vbCrLf & _
        "現行のみ → 現行 BP", _
        vbInformation, _
        "データ比較完了"
    Exit Sub
'====================================================================
' エラー処理
'====================================================================
ErrorHandler:
    Application.ScreenUpdating = prevScreenUpdating
    Application.EnableEvents = prevEnableEvents
    Application.Calculation = prevCalculation
    MsgBox _
        "データ比較中にエラーが発生しました。" & vbCrLf & vbCrLf & _
        "エラー番号：" & Err.Number & vbCrLf & _
        "エラー内容：" & Err.Description, _
        vbCritical, _
        "データ比較エラー"
End Sub
'====================================================================
' Key処理
'
' 空白Key：
'   空文字列を返す
'
' その他：
'   文字列に変換し、前後の空白を削除
'====================================================================
Private Function GetKey(ByVal v As Variant) As String
    If IsError(v) Then
        GetKey = ""
    ElseIf IsEmpty(v) Then
        GetKey = ""
    Else
        GetKey = Trim$(CStr(v))
    End If
End Function
'====================================================================
' 数値変換
'
' 空白      → 0
' 数値以外    → 0
' Error     → 0
' 数値      → Double
'====================================================================
Private Function ToNumberFast(ByVal v As Variant) As Double
    If IsError(v) Then
        ToNumberFast = 0
    ElseIf IsEmpty(v) Then
        ToNumberFast = 0
    ElseIf IsNumeric(v) Then
        ToNumberFast = CDbl(v)
    Else
        ToNumberFast = 0
    End If
End Function
'====================================================================
' 列番号からExcelの列記号を取得
'
' 11 → K
' 26 → Z
' 38 → AL
' 50 → AX
'====================================================================
Private Function ColumnLetter(ByVal colNum As Long) As String
    ColumnLetter = _
        Split( _
            Application.Cells(1, colNum).Address( _
                True, _
                False _
            ), _
            "$" _
        )(0)
End Function
